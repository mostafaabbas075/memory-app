    <h1>Memory App</h1>

    <p>
      Save something you want to remember.
    </p>

    <input
      id="knowledgeInput"
      type="text"
      placeholder="Enter your knowledge"
    >

    <button id="saveButton">
      Save
    </button>

    <p id="message"></p>

  </div>

  <script>
    const input = document.getElementById("knowledgeInput");
    const button = document.getElementById("saveButton");
    const message = document.getElementById("message");

    button.addEventListener("click", async function () {

      const knowledge = input.value.trim();

      if (!knowledge) {
        message.textContent = "Please enter some knowledge.";
        message.style.color = "#dc2626";
        return;
      }

      button.disabled = true;
      button.textContent = "Saving...";

      try {

        const response = await fetch(
          "http://localhost:8000/knowledge",
          {
            method: "POST",

            headers: {
              "Content-Type": "application/json"
            },

            body: JSON.stringify({
              content: knowledge
            })
          }
        );

        if (!response.ok) {
          throw new Error("Failed to save knowledge.");
        }

        message.textContent = "Knowledge saved successfully.";
        message.style.color = "#16a34a";

        input.value = "";

      } catch (error) {

        message.textContent = "Failed to save knowledge.";
        message.style.color = "#dc2626";

      } finally {

        button.disabled = false;
        button.textContent = "Save";

      }
    });
  </script>
